classdef TestLiftedHarmonicResponseCompute < matlab.unittest.TestCase
% TestLiftedHarmonicResponseCompute

% Unit tests for LiftedHarmonicResponseCompute.

    methods(Test)

        function testResponseBinderIsApplied(testCase)
            [sp, lifted, memberTable, spectralColumn] = ...
                TestLiftedHarmonicResponseCompute.fixture();

            obj = LiftedHarmonicResponseCompute( ...
                snapshotParameters = sp, ...
                liftedGainFunctions = lifted);

            binder = @(mask) @(X, f) ...
                trapz(f, abs(X), 1) ./ trapz(f, mask(:));

            responses = obj.compute( ...
                memberTable = memberTable, ...
                spectralColumn = spectralColumn, ...
                receptiveFieldResponseBinder = binder);

            f = lifted.frequencyDomain;

            X = TestLiftedHarmonicResponseCompute.stackSpectra( ...
                memberTable.(spectralColumn));

            tbl = lifted.liftedGainFunctionsTable;

            for i = 1:height(tbl)
                masks = tbl.FrequencyBankMasks{i};

                rfC = binder(masks.CenterMask(:));
                rfS = binder(masks.SurroundMask(:));

                nFFT = size(X, 1);

                expC = rfC((masks.CenterMask(:) .* X) ./ nFFT, f).';
                expS = rfS((masks.SurroundMask(:) .* X) ./ nFFT, f).';

                testCase.verifyEqual( ...
                    responses.Center(:, i), ...
                    expC, ...
                    "AbsTol", 1e-12);

                testCase.verifyEqual( ...
                    responses.Surround(:, i), ...
                    expS, ...
                    "AbsTol", 1e-12);
            end
        end

            [sp, lifted, memberTable, spectralColumn] = ...
                TestLiftedHarmonicResponseCompute.fixture();

            obj = LiftedHarmonicResponseCompute( ...
                snapshotParameters = sp, ...
                liftedGainFunctions = lifted);

            expectedF = lifted.frequencyDomain;

                TestLiftedHarmonicResponseCompute ...
                .verifyFrequencyAxisAndReturnZeros(X, f, expectedF);

            responses = obj.compute( ...
                memberTable = memberTable, ...
                spectralColumn = spectralColumn, ...
                receptiveFieldResponseFunction = rf);

            testCase.verifyEqual( ...
                responses.Center, ...
                zeros(size(responses.Center)));

            testCase.verifyEqual( ...
                responses.Surround, ...
                zeros(size(responses.Surround)));

    methods(Static, Access = private)

        function [sp, lifted, memberTable, spectralColumn] = fixture(args)
            arguments
                args.characteristicFrequencyHz (1,1) double = 50
                args.harmonicIndices (:,1) double = [1; 2; 4]
            end

            sp = ParametersSnapshot( ...
                samplingFrequency = 512, ...
                duration = 1, ...
                stride = 1);

            sharedParams = SFRFsParameters.buildSFRFsParameters( ...
                order = 0, ...
                numSidebands = 0, ...
                numHarmonics = 4, ...
                centerMask = SuperGaussianMaskParameters( ...
                    alpha = 2, ...
                    beta = 8), ...
                surroundMask = SuperGaussianMaskParameters( ...
                    alpha = 6, ...
                    beta = 2), ...
                inhibitionFactor = 0.9);

            sfrfsParams = SFRFsParametersRollingBearings( ...
                SameForAllFaultTypes = sharedParams);

            ft = SFRFsParametersRollingBearings ...
                .OUTER_RACE_FAULT_TYPE_NAME;

            sel = OperatingConditionSelection( ...
                speed = 35, ...
                load = 12);

            lifted = LiftedHarmonicGainFunctions( ...
                snapshotParameters = sp, ...
                sfrfsParameters = sfrfsParams, ...
                faultType = ft, ...
                selection = sel, ...
                characteristicFrequencyHz = ...
                    args.characteristicFrequencyHz, ...
                harmonicIndices = args.harmonicIndices);

            lifted.compute();

            f = sp.getFrequencyDomain();
            nSnapshots = 5;

            spectra = cell(nSnapshots, 1);

            for j = 1:nSnapshots
                phase = 2*pi*j*f ./ max(f + eps);

                spectra{j} = complex( ...
                    sin(phase), ...
                    cos(phase));
            end

            spectralColumn = "SyntheticFFT";

            memberTable = table( ...
                (1:nSnapshots).', ...
                spectra, ...
                'VariableNames', {'SnapshotIndex', char(spectralColumn)});
        end

        function X = stackSpectra(spectra)
            if ~iscell(spectra)
                X = spectra;
                return
            end

            nSnapshots = numel(spectra);
            nBins = numel(spectra{1});

            X = zeros(nBins, nSnapshots, "like", spectra{1});

            for i = 1:nSnapshots
                X(:, i) = spectra{i}(:);
            end
        end

        function r = verifyFrequencyAxisAndReturnZeros(X, f, expectedF)
            if ~isequal(f(:), expectedF(:))
                error( ...
                    "sfrfs:TestLiftedHarmonicResponseCompute:" + ...
                    "UnexpectedFrequencyAxis", ...
                    "Unexpected frequency axis.");
            end

            r = zeros(1, size(X, 2));
        end

    end
